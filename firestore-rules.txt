// Docket security rules — shared workspace with a private PIT lane + PIT store.
// Paste into: Firebase console > Firestore Database > Rules > Publish.

rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {
    function owner() {
      return request.auth != null
        && request.auth.token.email.lower() == 'griffgamble1@gmail.com';
    }
    function member() {
      return request.auth != null && request.auth.token.email.lower() in [
        'griffgamble1@gmail.com',
        'byhollyholiday@gmail.com',
        'hello@gm-charlie.com'
      ];
    }
    match /shared_board/main {
      allow read, write: if member();
    }
    match /private_board/pit {
      allow read, write: if owner();
    }
    match /docket/{uid} {
      allow read, write: if request.auth != null && request.auth.uid == uid;
    }

    // PIT store: products + orders — owner only (server uses admin SDK)
    match /pit_products/{productId} {
      allow read, write: if owner();
    }
    match /pit_orders/{orderId} {
      allow read, write: if owner();
    }
    match /pit_meta/{docId} {
      allow read, write: if owner();
    }
  }
}
